package your.package.storage

data class NovelMetadata(
    val id: String,
    val title: String,
    val sourceUri: String
)

data class ChapterIndex(
    val number: Int,
    val title: String,
    val startByte: Long,
    val endByte: Long
)

data class ReadingProgress(
    val chapterNumber: Int,
    val blockIndex: Int,
    val byteOffset: Long
)

data class NovelDocument(
    val blocks: List<NovelBlock>
)

sealed class NovelBlock {
    data class Heading(val level: Int, val title: String) : NovelBlock()
    data class Paragraph(
        val text: String,
        val fullHighlight: Boolean = false
    ) : NovelBlock()
}
